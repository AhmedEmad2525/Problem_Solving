-- problem 5
-- solution with subquery 
SELECT 
    machine_id,
    ROUND(AVG(end_date - start_date), 3) AS processing_time
FROM (
    SELECT 
        machine_id,
        process_id,
        MIN(CASE WHEN activity_type = 'start' THEN timestamp END) AS start_date, -- in case there is multi values return the minimum value
        MAX(CASE WHEN activity_type = 'end' THEN timestamp END) AS end_date -- in case there is multi values return the max value
    FROM Activity
    GROUP BY machine_id, process_id
) AS subquery
GROUP BY 
    machine_id;

-- solution with inner join
select a1.machine_id
, ROUND(AVG(a2.timestamp - a1.timestamp),3) as processing_time
from Activity a1
join Activity a2
on a1.process_id=a2.process_id
and a1.machine_id=a2.machine_id
and a1.timestamp<a2.timestamp
group by a1.machine_id;

-- inital solution 
select s.user_id,
COALESCE(
	ROUND(
	(
		 CAST(COUNT(CASE WHEN c.action = 'confirmed' THEN 1 ELSE NULL END) AS FLOAT)/
		 NULLIF(COUNT(CASE WHEN c.action IN ('confirmed','timeout') THEN 1  END),0)
	 )
	,2) 
,0) AS confirmation_rate

from signups s full outer join confirmations c
on s.user_id = c.user_id
group by s.user_id

-- better solution
select s.user_id,
COALESCE(
	ROUND(
	(
		 CAST(COUNT(CASE WHEN c.action = 'confirmed' THEN 1 ELSE NULL END) AS FLOAT)/
		 COUNT(*)
	 )
	,2) 
,0) AS confirmation_rate

from signups s left join confirmations c
on s.user_id = c.user_id
group by s.user_id

-- problem 4
-- solution using common table expression (CTE)
with Participate_percent AS(
select contest_id,
ROUND(
	(CAST(COUNT(r.user_id)AS float)/(select count(user_id) from users))*100
	,2)
 AS percentage
from Register r right join users u
on u.user_id = r.user_id
group by contest_id
)
select *
from Participate_percent
where contest_id is not null
ORDER by percentage DESC, contest_id

--  soluiton without CTE
SELECT
    contest_id,
    ROUND(COUNT(user_id) * 100.0 / (SELECT COUNT(DISTINCT user_id) FROM Users), 2) AS percentage
FROM
    Register
GROUP BY
    contest_id
ORDER BY
    percentage DESC

-- problem 5
-- quality = rating / position

with CTE as (
select query_name,

ROUND (
	AVG(CAST(rating as float)/NULLIF(position,0))
,2) as quality,

ROUND(
	CAST(COUNT(CASE when rating<3 then 1 END) as float)/NULLIF(COUNT(query_name),0) * 100
,2)AS poor_query_percentage 

from queries 
group by query_name
)
select * 
from CTE 
where poor_query_percentage is not null

-- simpler solution 
SELECT query_name, 
    ROUND(AVG(1.0 * rating / position), 2) AS quality,
    ROUND(SUM(IIF(rating < 3, 100.0, 0)) / COUNT(rating), 2) AS poor_query_percentage
FROM Queries
GROUP BY query_name
HAVING query_name IS NOT NULL;

-- problem 6
with date_month AS(
	select CONCAT(DATEPART(year,trans_date),'-',RIGHT('00' + CAST(DATEPART(month, trans_date) AS VARCHAR), 2)) AS month,*
	from Transactions
)
select month, country,
	COUNT(id) AS trans_count,
	COUNT(CASE when state = 'approved' then 1 END) AS approved_count,
	SUM(amount) AS trans_total_amount,
	SUM(CASE when state = 'approved' then amount ELSE 0 END) AS approved_total_amount
from 
	date_month
group by
	month, country
order by 
	month,country DESC

-- another solution using substring method for the date
with date_month AS(
     SELECT
	 SUBSTRING(CAST(trans_date as varchar), 1, 7) AS month,*
	from Transactions
)
SELECT 
    month,
    country,
    COUNT(id) AS trans_count,
    SUM(CASE WHEN state = 'approved' THEN 1 ELSE 0 END) AS approved_count,
    SUM(amount) AS trans_total_amount,
    SUM(CASE WHEN state = 'approved' THEN amount ELSE 0 END) AS approved_total_amount
FROM 
    date_month
GROUP BY 
    month, country;

-- problem 8

WITH CTE AS (
select 
	COUNT(a1.player_id) AS consecutive 
from 
	( -- get the first log day
		select player_id, MIN(event_date) AS event_date
		from Activity
		group by player_id
	) a1 JOIN Activity a2
	ON DateDIFF(day, a1.event_date , a2.event_date)=1 and a1.player_id = a2.player_id
)
select 
	ROUND(
	CAST(consecutive AS float)/(select COUNT(DISTINCT player_id) FROM Activity)
	,2) AS fraction
from 
	CTE

-- problem 1
-- solution using CTE (initial solution)
WITH manager AS( -- table to provide the report counts for each manager
	SELECT 
		reports_to AS employee_id, COUNT(reports_to) AS reports_count
	FROM
		Employees
	WHERE
		reports_to is not null
	GROUP BY
		reports_to
),
manager_ages AS( -- table to provide the average age for each manager
	SELECT m.employee_id ,ROUND(AVG(e.age*1.00),0) AS average_age
	FROM
		employees e left join manager m
	ON
		m.employee_id = e.reports_to
	WHERE
		m.employee_id IS NOT NULL
	GROUP BY
		m.employee_id
)
SELECT 
	m.employee_id, e.name, m.reports_count, ma.average_age
FROM 
	manager m INNER JOIN manager_ages ma
ON 
	m.employee_id = ma.employee_id
	INNER JOIN employees e
	ON	e.employee_id = m.employee_id
ORDER BY
    m.employee_id
-- solution using self join
SELECT
	e1.employee_id,e1.name,count(e2.reports_to) reports_count,round(avg(e2.age*1.00),0) average_age
FROM
	employees e1 INNER JOIN employees e2 
ON
	e1.employee_id = e2.reports_to
group by 
	e1.employee_id,e1.name
order by 
	e1.employee_id
